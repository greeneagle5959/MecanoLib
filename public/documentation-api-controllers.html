Base API: /api/v1 Format: JSON Date: 31/03/2026 Projet: MecanoLib

Format des routes: METHODE PATH. Les routes sans slash initial dans le code restent list ees ici en forme finale (avec /).

Comment utiliser ces routes dans React

1) Definis une base URL unique. 2) Centralise les appels API. 3) Ajoute le token JWT quand necessaire.

// src/api/client.js
const API_BASE = "http://localhost:8000";

export async function apiFetch(path, { method = "GET", body, token } = {}) {
  const res = await fetch(`${API_BASE}${path}`, {
    method,
    headers: {
      "Content-Type": "application/json",
      ...(token ? { Authorization: `Bearer ${token}` } : {}),
    },
    body: body ? JSON.stringify(body) : undefined,
  });

  const data = await res.json().catch(() => ({}));
  if (!res.ok) {
    throw new Error(data.message || data.error || `HTTP ${res.status}`);
  }
  return data;
}
// Exemple login + recuperation vehicules client
import { apiFetch } from "./api/client";

export async function login(emailUtilisateur, mdpUtilisateur) {
  return apiFetch("/api/v1/users/login", {
    method: "POST",
    body: { emailUtilisateur, mdpUtilisateur }
  });
}

export async function getVehiculesClient(idClient, token) {
  return apiFetch(`/api/v1/client/vehicules/${idClient}`, { token });
}
